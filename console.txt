// COMPLETE UNFRIEND TOOL - Manual + Auto Mode
(function createCompleteUnfriendTool() {
    console.log("🔧 Creating Complete Unfriend Tool (Manual + Auto)...");
    
    let threeDotButtons = [];
    let friendNames = [];
    let currentIndex = 0;
    let unfriendedCount = 0;
    let currentHighlight = null;
    let isAutoMode = false;
    let autoStop = false;
    
    // Find all 3-dot buttons
    const findThreeDotButtons = () => {
        const buttons = [];
        const threeDotSVGs = document.querySelectorAll('svg[viewBox="0 0 16 16"]');
        
        for (let svg of threeDotSVGs) {
            const path = svg.querySelector('path');
            if (path && path.getAttribute('d') && path.getAttribute('d').includes("M2 8a1.25")) {
                let clickable = svg.closest('div[role="button"]');
                if (!clickable) clickable = svg.parentElement;
                if (clickable && !buttons.includes(clickable)) {
                    buttons.push(clickable);
                }
            }
        }
        
        console.log(`📊 Found ${buttons.length} friends`);
        return buttons;
    };
    
    // Get friend name
    const getFriendName = (button) => {
        const container = button.closest('div[role="article"], div[data-pagelet="ProfileTabs"] div, div[class*="x1yztbdb"]');
        if (container) {
            const nameSpan = container.querySelector('span[class*="x1lliihq"]:not([class*="secondary"])');
            if (nameSpan && nameSpan.textContent.trim()) {
                return nameSpan.textContent.trim().substring(0, 35);
            }
            const allSpans = container.querySelectorAll('span');
            for (let span of allSpans) {
                const text = span.textContent.trim();
                if (text && text.length > 2 && text.length < 50 && 
                    !text.includes('Friend') && !text.includes('mutual')) {
                    return text.substring(0, 35);
                }
            }
        }
        return "Friend";
    };
    
    // Highlight friend
    const highlightFriend = (button) => {
        if (currentHighlight) {
            currentHighlight.style.outline = '';
            currentHighlight.style.backgroundColor = '';
        }
        
        const container = button.closest('div[role="article"], div[data-pagelet="ProfileTabs"] div, div[class*="x1yztbdb"]') || button.parentElement?.parentElement;
        
        if (container) {
            container.style.outline = '3px solid red';
            container.style.outlineOffset = '3px';
            container.style.backgroundColor = 'rgba(255, 0, 0, 0.15)';
            currentHighlight = container;
            container.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
    };
    
    // Click Unfriend from menu
    const clickUnfriendFromMenu = async () => {
        await wait(400);
        
        const unfriendSpan = Array.from(document.querySelectorAll('span[dir="auto"]')).find(
            span => span.textContent && 
                    span.textContent.toLowerCase().includes('unfriend')
        );
        
        if (unfriendSpan) {
            let clickable = unfriendSpan.closest('div[role="button"]');
            if (!clickable) clickable = unfriendSpan.closest('.x6s0dn4');
            if (!clickable) clickable = unfriendSpan.parentElement?.parentElement;
            
            if (clickable) {
                clickable.click();
                return true;
            }
        }
        
        const allElements = document.querySelectorAll('span, div');
        for (let el of allElements) {
            if (el.textContent && el.textContent.toLowerCase().includes('unfriend')) {
                let clickable = el.closest('div[role="button"]');
                if (clickable) {
                    clickable.click();
                    return true;
                }
                if (el.parentElement?.click) {
                    el.parentElement.click();
                    return true;
                }
            }
        }
        
        return false;
    };
    
    // Confirm unfriend
    const confirmUnfriend = async () => {
        await wait(500);
        
        const allButtons = document.querySelectorAll('div[role="button"], button');
        for (let btn of allButtons) {
            const text = btn.textContent?.toLowerCase() || '';
            if (text === 'confirm' || text.includes('confirm') || text === 'unfriend') {
                btn.click();
                return true;
            }
        }
        
        const dialog = document.querySelector('div[role="dialog"]');
        if (dialog) {
            const buttons = dialog.querySelectorAll('div[role="button"]');
            for (let btn of buttons) {
                if (btn.textContent?.toLowerCase().includes('confirm')) {
                    btn.click();
                    return true;
                }
            }
        }
        
        return false;
    };
    
    // Unfriend single person
    const unfriendSingle = async (index, isAuto = false) => {
        if (!threeDotButtons.length) {
            updateStatus("❌ No friends found! Click REFRESH first.", true);
            return false;
        }
        
        if (index >= threeDotButtons.length) {
            if (isAuto) {
                updateStatus("✅ All friends on this page unfriended! Scroll down and click REFRESH to load more.", false);
                stopAutoMode();
            } else {
                updateStatus("✅ All friends processed! Click REFRESH after scrolling.", false);
            }
            return false;
        }
        
        const button = threeDotButtons[index];
        const name = friendNames[index];
        
        updateStatus(`${isAuto ? '🤖 AUTO' : '🔴 MANUAL'}: Processing ${name} (${index + 1}/${threeDotButtons.length})`);
        
        try {
            button.scrollIntoView({ behavior: 'smooth', block: 'center' });
            await wait(400);
            button.click();
            await wait(500);
            
            const unfriendClicked = await clickUnfriendFromMenu();
            if (!unfriendClicked) {
                updateStatus(`⚠️ Could not find Unfriend for ${name}`, true);
                document.body.click();
                await wait(300);
                return false;
            }
            
            await wait(400);
            const confirmed = await confirmUnfriend();
            
            if (confirmed) {
                unfriendedCount++;
                updateStatus(`✅ ${name} unfriended! (Total: ${unfriendedCount})`);
                markAsUnfriended(index);
                
                if (currentHighlight) {
                    currentHighlight.style.outline = '';
                    currentHighlight.style.backgroundColor = '';
                    currentHighlight = null;
                }
                
                await wait(600);
                return true;
            }
            
            updateStatus(`❌ Failed to confirm for ${name}`, true);
            document.body.click();
            return false;
            
        } catch (err) {
            updateStatus(`❌ Error: ${err.message}`, true);
            return false;
        }
    };
    
    // Auto mode - unfriend all continuously
    const startAutoMode = async () => {
        if (isAutoMode) {
            stopAutoMode();
            return;
        }
        
        isAutoMode = true;
        autoStop = false;
        updateAutoButton(true);
        updateStatus("🤖 AUTO MODE STARTED - Unfriending all friends automatically...");
        
        while (isAutoMode && !autoStop && currentIndex < threeDotButtons.length) {
            const success = await unfriendSingle(currentIndex, true);
            
            if (success) {
                currentIndex++;
                updateProgress();
            } else {
                // If failed, try next or stop
                currentIndex++;
                updateStatus(`⚠️ Skipping to next...`, true);
                await wait(500);
            }
            
            // Update progress
            updateProgress();
            
            // If all done on this page
            if (currentIndex >= threeDotButtons.length) {
                updateStatus("✅ All friends on this page done! Scroll manually and click REFRESH to load more, then start AUTO again.", false);
                stopAutoMode();
                break;
            }
            
            // Highlight next if in manual mode preview
            if (!isAutoMode) break;
        }
        
        if (isAutoMode) {
            stopAutoMode();
        }
    };
    
    const stopAutoMode = () => {
        isAutoMode = false;
        autoStop = true;
        updateAutoButton(false);
        updateStatus("⏹️ AUTO MODE STOPPED", false);
    };
    
    // Mark unfriended in list
    const markAsUnfriended = (index) => {
        const listItem = document.getElementById(`friend_${index}`);
        if (listItem) {
            listItem.style.textDecoration = 'line-through';
            listItem.style.color = '#8a8d91';
            listItem.style.backgroundColor = '#e8f5e9';
        }
    };
    
    // Refresh list
    const refreshList = () => {
        threeDotButtons = findThreeDotButtons();
        friendNames = threeDotButtons.map(btn => getFriendName(btn));
        currentIndex = 0;
        unfriendedCount = 0;
        
        if (currentHighlight) {
            currentHighlight.style.outline = '';
            currentHighlight.style.backgroundColor = '';
            currentHighlight = null;
        }
        
        updateFriendListDisplay();
        
        if (threeDotButtons.length > 0) {
            updateStatus(`✅ Found ${threeDotButtons.length} friends! Use MANUAL (step-by-step) or AUTO mode.`);
            highlightFriend(threeDotButtons[0]);
        } else {
            updateStatus("❌ No friends found! Make sure you're on the Facebook Friends page.", true);
        }
        
        updateProgress();
    };
    
    // Manual next
    const manualNext = async () => {
        if (isAutoMode) {
            updateStatus("⚠️ Please stop AUTO mode first by clicking the AUTO button again.", true);
            return;
        }
        
        if (currentIndex >= threeDotButtons.length) {
            updateStatus("✅ All done! Click REFRESH after scrolling to load more friends.", false);
            return;
        }
        
        // Highlight current before unfriending
        if (currentIndex < threeDotButtons.length) {
            highlightFriend(threeDotButtons[currentIndex]);
        }
        
        const success = await unfriendSingle(currentIndex, false);
        
        if (success) {
            currentIndex++;
            updateProgress();
            
            // Highlight next
            if (currentIndex < threeDotButtons.length) {
                highlightFriend(threeDotButtons[currentIndex]);
            } else {
                if (currentHighlight) {
                    currentHighlight.style.outline = '';
                    currentHighlight.style.backgroundColor = '';
                    currentHighlight = null;
                }
                updateStatus("🎉 All friends on this page unfriended! Scroll down and click REFRESH.", false);
            }
        }
    };
    
    const updateProgress = () => {
        const progress = (currentIndex / threeDotButtons.length) * 100;
        document.getElementById('progressBar').style.width = progress + '%';
        document.getElementById('unfriendedCount').textContent = unfriendedCount;
        document.getElementById('remainingCount').textContent = threeDotButtons.length - currentIndex;
        document.getElementById('totalCount').textContent = threeDotButtons.length;
    };
    
    const updateAutoButton = (isRunning) => {
        const autoBtn = document.getElementById('autoModeBtn');
        if (autoBtn) {
            if (isRunning) {
                autoBtn.textContent = '⏹️ STOP AUTO MODE';
                autoBtn.style.background = '#ff9800';
            } else {
                autoBtn.textContent = '🤖 AUTO MODE (Unfriend All)';
                autoBtn.style.background = '#4caf50';
            }
        }
    };
    
    const updateFriendListDisplay = () => {
        const listContainer = document.getElementById('friendListContainer');
        if (!listContainer) return;
        
        if (threeDotButtons.length === 0) {
            listContainer.innerHTML = '<div style="color: #e41e3f; text-align: center; padding: 20px;">❌ No friends found. Go to Facebook Friends page and click REFRESH.</div>';
            return;
        }
        
        let html = '<ul style="margin: 0; padding-left: 20px; max-height: 250px; overflow-y: auto;">';
        friendNames.forEach((name, i) => {
            html += `<li id="friend_${i}" style="margin: 5px 0; padding: 4px; border-radius: 4px;">${i+1}. ${name}</li>`;
        });
        html += '</ul>';
        listContainer.innerHTML = html;
    };
    
    const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms));
    
    const updateStatus = (message, isError = false) => {
        const statusDiv = document.getElementById('statusMessage');
        if (statusDiv) {
            statusDiv.innerHTML = message;
            statusDiv.style.color = isError ? '#e41e3f' : '#1877f2';
        }
        console.log(message);
    };
    
    // Create UI Panel
    const panel = document.createElement('div');
    panel.style.cssText = `
        position: fixed;
        bottom: 20px;
        right: 20px;
        width: 420px;
        background: white;
        border-radius: 12px;
        box-shadow: 0 8px 28px rgba(0,0,0,0.3);
        font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
        z-index: 10000;
        overflow: hidden;
        border: 1px solid #ddd;
    `;
    
    panel.innerHTML = `
        <div style="background: linear-gradient(135deg, #667eea 0%, #764ba2 100%); color: white; padding: 12px 16px; font-weight: bold; display: flex; justify-content: space-between; align-items: center;">
            <span>🎯 UNFRIEND TOOL v2.0</span>
            <button id="closePanelBtn" style="background: none; border: none; color: white; font-size: 20px; cursor: pointer;">✕</button>
        </div>
        <div style="padding: 16px;">
            <!-- Stats -->
            <div style="background: #f0f2f5; border-radius: 8px; padding: 12px; margin-bottom: 12px;">
                <div style="display: flex; justify-content: space-between; margin-bottom: 8px;">
                    <span><strong>✅ Unfriended:</strong> <span id="unfriendedCount">0</span></span>
                    <span><strong>📊 Total:</strong> <span id="totalCount">0</span></span>
                    <span><strong>👥 Remaining:</strong> <span id="remainingCount">0</span></span>
                </div>
                <div style="height: 8px; background: #ddd; border-radius: 4px; overflow: hidden;">
                    <div id="progressBar" style="height: 100%; background: #4caf50; width: 0%; transition: width 0.3s;"></div>
                </div>
            </div>
            
            <!-- Friend List -->
            <div id="friendListContainer" style="background: #f0f2f5; border-radius: 8px; padding: 8px; margin-bottom: 12px; max-height: 250px; overflow-y: auto;">
                Click REFRESH to load friends
            </div>
            
            <!-- Control Buttons -->
            <div style="display: flex; gap: 10px; margin-bottom: 12px;">
                <button id="manualBtn" style="flex: 1; background: #e41e3f; color: white; border: none; padding: 12px; border-radius: 8px; cursor: pointer; font-weight: bold; font-size: 14px;">
                    🔴 MANUAL (1 by 1)
                </button>
                <button id="autoModeBtn" style="flex: 1; background: #4caf50; color: white; border: none; padding: 12px; border-radius: 8px; cursor: pointer; font-weight: bold; font-size: 14px;">
                    🤖 AUTO MODE
                </button>
                <button id="refreshBtn" style="flex: 0.7; background: #1877f2; color: white; border: none; padding: 12px; border-radius: 8px; cursor: pointer; font-weight: bold;">
                    🔄 REFRESH
                </button>
            </div>
            
            <!-- Status -->
            <div id="statusMessage" style="font-size: 12px; padding: 8px; background: #f0f2f5; border-radius: 6px; color: #65676b;">
                🔍 Click REFRESH to find friends, then choose MANUAL or AUTO mode.
            </div>
            
            <!-- Instructions -->
            <div style="font-size: 11px; margin-top: 8px; padding: 6px; background: #fff3cd; border-radius: 6px; color: #856404;">
                <strong>📖 How to use:</strong><br>
                • <strong>REFRESH</strong> - Load friends from current page<br>
                • <strong>MANUAL</strong> - Unfriend ONE person (shows RED highlight)<br>
                • <strong>AUTO MODE</strong> - Unfriends ALL automatically (click again to stop)<br>
                • Scroll down manually → Click REFRESH → Continue
            </div>
        </div>
    `;
    
    document.body.appendChild(panel);
    
    // Event listeners
    document.getElementById('manualBtn').addEventListener('click', async () => {
        if (isAutoMode) {
            updateStatus("⚠️ Please stop AUTO mode first by clicking the AUTO button again.", true);
            return;
        }
        await manualNext();
    });
    
    document.getElementById('autoModeBtn').addEventListener('click', () => {
        if (isAutoMode) {
            stopAutoMode();
        } else {
            startAutoMode();
        }
    });
    
    document.getElementById('refreshBtn').addEventListener('click', () => {
        if (isAutoMode) {
            stopAutoMode();
        }
        refreshList();
    });
    
    document.getElementById('closePanelBtn').addEventListener('click', () => {
        if (isAutoMode) stopAutoMode();
        if (currentHighlight) {
            currentHighlight.style.outline = '';
            currentHighlight.style.backgroundColor = '';
        }
        panel.remove();
    });
    
    // Initial load
    refreshList();
    
    console.log(`
╔═══════════════════════════════════════════╗
║   🎯 COMPLETE UNFRIEND TOOL READY!        ║
╠═══════════════════════════════════════════╣
║  • REFRESH - Load friends                 ║
║  • MANUAL - Unfriend 1 person with RED    ║
║  • AUTO - Unfriends ALL automatically     ║
║  • Click AUTO again to stop               ║
╚═══════════════════════════════════════════╝
    `);
})();
